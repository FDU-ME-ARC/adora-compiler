# ADORA cycle-estimator: an importable Python package.
#
# Installed as a Python package tree under <prefix>/bin/pypack/cycle_estimator
# so it can be invoked as:
#     PYTHONPATH=<prefix>/bin/pypack python -m cycle_estimator --mlir k.mlir
#
# The estimator reads loop trip-counts strong-typed via the upstream MLIR
# Python bindings (mlir_core) + the bundled adora_mlir dialect package; set
# ADORA_MLIR_CORE to the LLVM build's python_packages/mlir_core at runtime.

set(_CE_DEST ${CMAKE_INSTALL_PREFIX}/bin/pypack/cycle_estimator)

install(DIRECTORY
          ${CMAKE_CURRENT_SOURCE_DIR}/core
          ${CMAKE_CURRENT_SOURCE_DIR}/extract
          ${CMAKE_CURRENT_SOURCE_DIR}/arch
          ${CMAKE_CURRENT_SOURCE_DIR}/scripts
          ${CMAKE_CURRENT_SOURCE_DIR}/adora_mlir
        DESTINATION ${_CE_DEST}
        FILES_MATCHING
          PATTERN "*.py"
          PATTERN "*.so"
          PATTERN "*.json"
          PATTERN "__pycache__" EXCLUDE)

install(FILES
          ${CMAKE_CURRENT_SOURCE_DIR}/run.py
        DESTINATION ${_CE_DEST})

# Expose the package as `cycle_estimator` with a __main__ entry so that
# `python -m cycle_estimator` runs the estimator.
install(CODE "file(WRITE \"${_CE_DEST}/__init__.py\" \"\")")
# __main__ puts the package dir on sys.path so run.py's `from core.*` and the
# `from run import main` below both resolve under `python -m cycle_estimator`.
install(CODE "file(WRITE \"${_CE_DEST}/__main__.py\" \"import os, sys\\nsys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))\\nfrom run import main\\nmain()\\n\")")

# Generate a bin/ launcher so users can run `cycle-estimator ...` directly
# without setting PYTHONPATH. It points PYTHONPATH at the installed pypack and
# provides a default ADORA_MLIR_CORE (overridable by the environment).
set(_CE_MLIR_CORE_DEFAULT
    "/data00/home/loujiahang/CGRVOPT/llvm-project-onnx/build/python_packages/mlir_core")
set(_CE_LAUNCHER ${CMAKE_CURRENT_BINARY_DIR}/cycle-estimator)
file(WRITE ${_CE_LAUNCHER}
"#!/usr/bin/env bash
# ADORA cycle-estimator launcher (generated by CMake).
export PYTHONPATH=\"${CMAKE_INSTALL_PREFIX}/bin/pypack:\${PYTHONPATH}\"
export ADORA_MLIR_CORE=\"\${ADORA_MLIR_CORE:-${_CE_MLIR_CORE_DEFAULT}}\"
exec python3 -m cycle_estimator \"\$@\"
")
install(PROGRAMS ${_CE_LAUNCHER} DESTINATION ${CMAKE_INSTALL_PREFIX}/bin)
